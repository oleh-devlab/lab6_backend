# GET /health — 200 (база доступна)
curl -X 'GET' \
  'http://localhost:8080/health' \
  -H 'accept: application/json'

# POST /api/links — 201 (код генерує сервіс)
curl -X 'POST' \
  'http://localhost:8080/api/links' \
  -H 'accept: application/json' \
  -H 'Content-Type: application/json' \
  -d '{
  "url": "https://expressjs.com/en/5x/api.html"
}'

# POST /api/links — 201 (власний alias)
curl -X 'POST' \
  'http://localhost:8080/api/links' \
  -H 'accept: application/json' \
  -H 'Content-Type: application/json' \
  -d '{
  "url": "https://expressjs.com/en/5x/api.html",
  "alias": "express-api"
}'

# POST /api/links — 409 (alias уже зайнятий)
curl -X 'POST' \
  'http://localhost:8080/api/links' \
  -H 'accept: application/json' \
  -H 'Content-Type: application/json' \
  -d '{
  "url": "https://nodejs.org/",
  "alias": "express-api"
}'

# POST /api/links — 400 (не http/https-адреса)
curl -X 'POST' \
  'http://localhost:8080/api/links' \
  -H 'accept: application/json' \
  -H 'Content-Type: application/json' \
  -d '{
  "url": "javascript:alert(1)"
}'

# POST /api/links — 400 (зарезервований alias)

  -H 'accept: application/json' \
  -H 'Content-Type: application/json' \
  -d '{
  "url": "https://nodejs.org/",

}'

# GET /api/links — 200
curl -X 'GET' \
  'http://localhost:8080/api/links?limit=20&offset=0' \
  -H 'accept: application/json'

# GET /api/links — 400 (limit поза межами)
curl -X 'GET' \
  'http://localhost:8080/api/links?limit=500&offset=0' \
  -H 'accept: application/json'

# GET /api/links/{code} — 200
curl -X 'GET' \
